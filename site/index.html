bash — custom domain deploy 
git clone https://github.com/pablobravo44/pulse-analytics-cloudflare.git
cd pulse-analytics-cloudflare
npm ci && npx wrangler login

npx wrangler d1 create pulse --jurisdiction eu
npx wrangler queues create pulse-events
npx wrangler queues create pulse-events-dlq
npx wrangler kv namespace create OAUTH_KV
npx wrangler r2 bucket create pulse-backups --jurisdiction eu

cp .env.deploy.example .env.deploy   # fill PULSE_DOMAIN, PULSE_D1_ID, PULSE_KV_ID
npm run build && npm run deploy:custom
